func _initialize() -> void:
	if not "--write-island-v4" in OS.get_cmdline_user_args() or FileAccess.get_sha256(BACKUP) != BACKUP_SHA256:
		push_error("Rebuild requires -- --write-island-v4 and the unchanged Task 7.3 native backup. Later sculpting WILL be replaced.")
		quit(1)
		return
	_build.call_deferred()


func _outline(points: Array) -> PackedVector2Array:
	var result := PackedVector2Array()
	for i in points.size():
		var a: Vector2 = points[posmod(i - 1, points.size())]
		var b: Vector2 = points[i]
		var c: Vector2 = points[(i + 1) % points.size()]
		var d: Vector2 = points[(i + 2) % points.size()]
		for step in 6:
			var t := step / 6.0
			result.append(0.5 * ((2 * b) + (-a + c) * t + (2 * a - 5 * b + 4 * c - d) * t * t + (-a + 3 * b - 3 * c + d) * t * t * t))
	return result


func _signed_distance(p: Vector2, polygon: PackedVector2Array) -> float:
	var distance := INF
	for i in polygon.size():
		var a := polygon[i]
		var segment := polygon[(i + 1) % polygon.size()] - a
		var t := clampf((p - a).dot(segment) / segment.length_squared(), 0, 1)
		distance = minf(distance, p.distance_squared_to(a + segment * t))
	return sqrt(distance) * (1 if Geometry2D.is_point_in_polygon(p, polygon) else -1)


func _shore(x: int, z: int) -> float:
	var ix := x / 4
	var iz := z / 4
	return lerpf(lerpf(coast_field[iz * FIELD_SIZE + ix], coast_field[iz * FIELD_SIZE + ix + 1], (x % 4) / 4.0), lerpf(coast_field[(iz + 1) * FIELD_SIZE + ix], coast_field[(iz + 1) * FIELD_SIZE + ix + 1], (x % 4) / 4.0), (z % 4) / 4.0)


func _ridge(p: Vector2, points: Array, width: float) -> float:
	var result := 0.0
	for i in points.size() - 1:
		var a: Vector3 = points[i]
		var b: Vector3 = points[i + 1]
		var start := Vector2(a.x, a.z)
		var segment := Vector2(b.x, b.z) - start
		var t := clampf((p - start).dot(segment) / segment.length_squared(), 0, 1)
		var distance := p.distance_to(start + segment * t)
		# Sharp crest with long convex flanks ending tangentially in the lowlands.
		result = maxf(result, lerpf(a.y, b.y, t) * pow(maxf(0, 1 - distance / width), 1.85))
	return result


func _hill(p: Vector2, center: Vector2, radii: Vector2, height: float) -> float:
	return height * exp(-((p - center) / radii).length_squared() * 1.8)


func _height(x: int, z: int) -> float:
	var p := Vector2(x, z)
	var shore := _shore(x, z)
	if shore < -45:
		return -8.0
	# Authored coastal sectors give narrow northern/eastern beaches and broad
	# western/southern bays, rather than a uniform sandy ribbon around the map.
	var beach_width := 48.0 + _hill(p, Vector2(265, 780), Vector2(160, 130), 48)
	beach_width += _hill(p, Vector2(130, 500), Vector2(100, 160), 35)
	beach_width += _hill(p, Vector2(510, 868), Vector2(150, 60), 30)
	var base := 9.0 * smoothstep(0, beach_width, shore) if shore >= 0 else -8.0 * smoothstep(0, 45, -shore)
	var cliff := maxf(_hill(p, Vector2(805, 265), Vector2(42, 46), 1), _hill(p, Vector2(155, 320), Vector2(34, 40), 1))
	cliff = maxf(cliff, _hill(p, Vector2(874, 651), Vector2(35, 45), 1))
	base = lerpf(base, lerpf(-8, 16, smoothstep(-8, 28, shore)), cliff)
	var inland := smoothstep(0, 65, shore)
	var warped := p + Vector2(noise.get_noise_2d(x, z), noise.get_noise_2d(x + 1234, z)) * 4.0
	var west := maxf(_ridge(warped, WEST, 142), _ridge(warped, WEST_OUTER, 95))
	west = maxf(west, maxf(_ridge(warped, WEST_INNER, 96), _ridge(warped, WEST_SOUTH, 85)))
	var east := maxf(_ridge(warped, EAST, 132), maxf(_ridge(warped, EAST_INNER, 92), _ridge(warped, EAST_OUTER, 88)))
	var mountains := maxf(west, east)
	# Relief belongs to mountain faces, never to the entire valley floor.
	mountains *= 1.0 + noise.get_noise_2d(x * 3, z * 3) * 0.045
	var hills := _hill(p, Vector2(302, 657), Vector2(100, 115), 11)
	hills += _hill(p, Vector2(720, 747), Vector2(90, 70), 7)
	hills += _hill(p, Vector2(575, 325), Vector2(110, 75), 7)
	hills += _hill(p, Vector2(416, 650), Vector2(52, 90), 4)
	hills += _hill(p, Vector2(450, 600), Vector2(65, 90), 9)
	hills += _hill(p, Vector2(625, 670), Vector2(90, 58), 14)
	hills += _hill(p, Vector2(610, 770), Vector2(65, 75), 8)
	hills += _hill(p, Vector2(460, 385), Vector2(80, 65), 7)
	hills += _hill(p, Vector2(230, 560), Vector2(75, 50), 13)
	hills += _hill(p, Vector2(730, 300), Vector2(65, 95), 17)
	var h := base + (mountains + _ridge(p, NORTH, 130) + hills) * inland
	if x > 400 and x < 660 and z > 400 and z < 650:
		var d := _signed_distance(p, lake_polygon)
		var basin := 5.0 - 3.2 * smoothstep(0, 13, d) if d >= 0 else 5.0 + 4.0 * smoothstep(0, 35, -d)
		var blend := 1.0 - smoothstep(0, 65, -d)
		var islet := ((p - Vector2(538, 524)) / Vector2(8, 5.5)).length()
		h = lerpf(h, basin + 6.7 * exp(-pow(islet, 3)), blend)
	return h


func _build() -> void:
	noise.seed = 74019
	noise.frequency = 0.018
	noise.fractal_octaves = 2
	var polygon := _outline(COAST)
	lake_polygon = _outline(LAKE)
	coast_field.resize(FIELD_SIZE * FIELD_SIZE)
	for z in FIELD_SIZE:
		for x in FIELD_SIZE:
			coast_field[z * FIELD_SIZE + x] = _signed_distance(Vector2(x * 4, z * 4), polygon)
	print("V4 coastline authored; shaping ridges, lowlands and lake")
	var height := Image.create_empty(SIZE, SIZE, false, Image.FORMAT_RF)
	var colors := Image.create_empty(SIZE, SIZE, false, Image.FORMAT_RGBA8)
	for z in SIZE:
		for x in SIZE:
			height.set_pixel(x, z, Color(_height(x, z), 0, 0))
			var shade := 0.96 + noise.get_noise_2d(x, z) * 0.035
			colors.set_pixel(x, z, Color(shade, shade, shade, 0.5))
	var world := Node3D.new()
	root.add_child(world)
	var camera := Camera3D.new()
	world.add_child(camera)
	camera.position = Vector3(500, 60, 650)
	camera.current = true
	var terrain := Terrain3D.new()
	world.add_child(terrain)
	terrain.set_camera(camera)
	terrain.region_size = Terrain3D.SIZE_1024
	terrain.data.import_images([height, null, colors], Vector3.ZERO, 0, 1)
	for z in SIZE:
		for x in SIZE:
			var h := height.get_pixel(x, z).r
			var slope := Vector2(height.get_pixel(mini(x + 1, SIZE - 1), z).r - height.get_pixel(maxi(x - 1, 0), z).r, height.get_pixel(x, mini(z + 1, SIZE - 1)).r - height.get_pixel(x, maxi(z - 1, 0)).r).length() * 0.5
			var sand := 1.0 - smoothstep(1.8, 5.0, h)
			if x > 455 and x < 615 and z > 450 and z < 600:
				sand = maxf(sand, (1.0 - smoothstep(5.0, 6.7, h)) * 0.9)
			var rock := maxf(smoothstep(0.46, 1.1, slope), smoothstep(42, 76, h) * 0.8) * smoothstep(3, 9, h)
			var point := Vector3(x, 0, z)
			terrain.data.set_control_base_id(point, 0)
			terrain.data.set_control_overlay_id(point, 2 if rock > 0.05 else 1)
			terrain.data.set_control_blend(point, rock if rock > 0.05 else sand)
	terrain.data.update_maps()
	terrain.data.save_directory(DIRECTORY + "/data")
	print("ISLAND V4 SAVED: regions=", terrain.data.get_region_count(), " range=", terrain.data.get_height_range(), " spawn=", terrain.data.get_height(Vector3(500, 0, 650)))
	world.queue_free()
	await process_frame
	quit()
